name: Build IPA
on:
  workflow_dispatch:
  push:
    branches: [main]
jobs:
  build:
    runs-on: macos-latest
    steps:
      - uses: actions/checkout@v4
      - name: Install XcodeGen
        run: brew install xcodegen
      - name: Generate Xcode project
        run: xcodegen generate
      - name: Build (unsigned)
        run: |
          xcodebuild -project PhaseLauncher.xcodeproj -scheme PhaseLauncher \
            -configuration Release -sdk iphoneos -derivedDataPath build \
            CODE_SIGNING_ALLOWED=NO CODE_SIGNING_REQUIRED=NO CODE_SIGN_IDENTITY="" build
      - name: Package IPA
        run: |
          mkdir Payload
          cp -R build/Build/Products/Release-iphoneos/PhaseLauncher.app Payload/
          zip -r PhaseLauncher.ipa Payload
      - uses: actions/upload-artifact@v4
        with:
          name: PhaseLauncher-ipa
          path: PhaseLauncher.ipa
